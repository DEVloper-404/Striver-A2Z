# Design Twitter

**Difficulty:** Medium  ·  **Topic:** Heaps [Learning, Medium, Hard Problems] › Hard Problems

## Problem
Implement a tiny social network with four operations:

- `postTweet(userId, tweetId)` – the user publishes a tweet.
- `getNewsFeed(userId)` – return the ids of the 10 most recent tweets written by the user or by anyone they follow, newest first.
- `follow(followerId, followeeId)` and `unfollow(followerId, followeeId)`.

Users may follow and unfollow freely; a user always sees their own tweets.

```
Input:  postTweet(1, 5); getNewsFeed(1); follow(1, 2); postTweet(2, 6);
        getNewsFeed(1); unfollow(1, 2); getNewsFeed(1)
Output: [5]; [6, 5]; [5]
```

## Approach 1: Global tweet log
Store every tweet in one list in posting order with its author. For `getNewsFeed`, walk the list backwards and collect tweets whose author is the user or someone they follow, stopping after 10. Simple, but a feed request scans through all tweets ever posted in the worst case.
- **Time:** O(1) per post/follow, O(total tweets) per feed
- **Space:** O(total tweets + follow edges)

## Approach 2: Per-user tweet lists merged with a heap (optimal)
Keep, for each user, their own list of `(timestamp, tweetId)` in posting order, and a set of people they follow. Building a feed is then the classic "merge k sorted lists, take the first 10":

1. Collect the users to read from: the requester plus their followees.
2. Push the most recent tweet of each of those users into a max-heap keyed by timestamp, together with the index of that tweet in the author's list.
3. Pop the newest tweet, append it to the result, and push the author's previous tweet (if any). Stop after 10 pops or when the heap is empty.

Only the frontier of each followed list lives in the heap, so its size is bounded by the number of followees `f`.
- **Time:** O(1) post/follow/unfollow; O(f + 10 log f) per feed
- **Space:** O(total tweets + follow edges)

## Code (Optimal)

### C++
```cpp
#include <bits/stdc++.h>
using namespace std;

class Twitter {
    int clock = 0;
    unordered_map<int, vector<pair<int, int>>> tweets;   // user -> [(time, tweetId)]
    unordered_map<int, unordered_set<int>> following;    // user -> followees

public:
    Twitter() {}

    void postTweet(int userId, int tweetId) {
        tweets[userId].push_back({clock++, tweetId});
    }

    vector<int> getNewsFeed(int userId) {
        // (time, tweetId, authorId, index in author's list)
        priority_queue<tuple<int, int, int, int>> maxHeap;

        auto pushLatest = [&](int author) {
            auto it = tweets.find(author);
            if (it == tweets.end() || it->second.empty()) return;
            int idx = it->second.size() - 1;
            maxHeap.push({it->second[idx].first, it->second[idx].second, author, idx});
        };

        pushLatest(userId);
        for (int followee : following[userId]) {
            if (followee != userId) pushLatest(followee);
        }

        vector<int> feed;
        while (!maxHeap.empty() && feed.size() < 10) {
            auto [time, tweetId, author, idx] = maxHeap.top();
            maxHeap.pop();
            feed.push_back(tweetId);
            if (idx > 0) {
                auto& list = tweets[author];
                maxHeap.push({list[idx - 1].first, list[idx - 1].second, author, idx - 1});
            }
        }
        return feed;
    }

    void follow(int followerId, int followeeId) {
        following[followerId].insert(followeeId);
    }

    void unfollow(int followerId, int followeeId) {
        following[followerId].erase(followeeId);
    }
};
```

### Java
```java
import java.util.*;

class Twitter {
    private int clock = 0;
    private final Map<Integer, List<int[]>> tweets = new HashMap<>();      // user -> [(time, tweetId)]
    private final Map<Integer, Set<Integer>> following = new HashMap<>();  // user -> followees

    public Twitter() {}

    public void postTweet(int userId, int tweetId) {
        tweets.computeIfAbsent(userId, k -> new ArrayList<>()).add(new int[]{clock++, tweetId});
    }

    public List<Integer> getNewsFeed(int userId) {
        // entries: {time, tweetId, authorId, index}
        PriorityQueue<int[]> maxHeap = new PriorityQueue<>((a, b) -> b[0] - a[0]);

        Set<Integer> sources = new HashSet<>(following.getOrDefault(userId, Collections.emptySet()));
        sources.add(userId);
        for (int author : sources) {
            List<int[]> list = tweets.get(author);
            if (list == null || list.isEmpty()) continue;
            int idx = list.size() - 1;
            maxHeap.offer(new int[]{list.get(idx)[0], list.get(idx)[1], author, idx});
        }

        List<Integer> feed = new ArrayList<>();
        while (!maxHeap.isEmpty() && feed.size() < 10) {
            int[] top = maxHeap.poll();
            feed.add(top[1]);
            int author = top[2], idx = top[3];
            if (idx > 0) {
                int[] prev = tweets.get(author).get(idx - 1);
                maxHeap.offer(new int[]{prev[0], prev[1], author, idx - 1});
            }
        }
        return feed;
    }

    public void follow(int followerId, int followeeId) {
        following.computeIfAbsent(followerId, k -> new HashSet<>()).add(followeeId);
    }

    public void unfollow(int followerId, int followeeId) {
        Set<Integer> s = following.get(followerId);
        if (s != null) s.remove(followeeId);
    }
}
```

### Python
```python
import heapq
from collections import defaultdict
from typing import List


class Twitter:
    def __init__(self):
        self.clock = 0
        self.tweets = defaultdict(list)       # user -> [(time, tweetId)]
        self.following = defaultdict(set)     # user -> set of followees

    def postTweet(self, userId: int, tweetId: int) -> None:
        self.tweets[userId].append((self.clock, tweetId))
        self.clock += 1

    def getNewsFeed(self, userId: int) -> List[int]:
        heap = []  # (-time, tweetId, author, index)
        sources = set(self.following[userId])
        sources.add(userId)
        for author in sources:
            lst = self.tweets.get(author)
            if lst:
                idx = len(lst) - 1
                heap.append((-lst[idx][0], lst[idx][1], author, idx))
        heapq.heapify(heap)

        feed = []
        while heap and len(feed) < 10:
            _, tweet_id, author, idx = heapq.heappop(heap)
            feed.append(tweet_id)
            if idx > 0:
                t, tid = self.tweets[author][idx - 1]
                heapq.heappush(heap, (-t, tid, author, idx - 1))
        return feed

    def follow(self, followerId: int, followeeId: int) -> None:
        self.following[followerId].add(followeeId)

    def unfollow(self, followerId: int, followeeId: int) -> None:
        self.following[followerId].discard(followeeId)
```

## Key Takeaways
- A global counter as the timestamp is enough; you never need real time, just a strict order.
- The feed is "merge k sorted lists" in disguise: push one frontier tweet per author, pop the newest, refill from that author.
- Always include the requester in the set of sources, and guard against a user following themselves so their tweets are not pushed twice.
- Stop after 10 results; do not merge every tweet and then slice, which would defeat the point of the heap.
