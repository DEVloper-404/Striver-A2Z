# LFU Cache

**Difficulty:** Hard  ·  **Topic:** Stack and Queues [Learning, Pre-In-Post-fix, Monotonic Stack, Implementation] › Implementation Problems

## Problem
Design a cache of fixed `capacity` with `get(key)` and `put(key, value)` in O(1). Each key has a *use counter* that increases on every `get` or `put` of that key (a fresh insert starts at 1). When the cache is full and a new key arrives, evict the key with the smallest use counter; if several keys tie, evict the *least recently used* among them.

```
Input:  capacity = 2
        put(1,1), put(2,2), get(1), put(3,3), get(2), get(3), put(4,4), get(1), get(3), get(4)
Output: get(1)=1, get(2)=-1, get(3)=3, get(1)=-1, get(3)=3, get(4)=4
```
When 3 is inserted, key 2 (freq 1) loses to key 1 (freq 2). When 4 is inserted, keys 1 and 3 both have freq 2, so the older one (1) is evicted.

## Approach 1: Scan for the Victim
Store `(value, freq, lastUsed)` per key in a hash map. On eviction scan all entries for the minimum `(freq, lastUsed)`.
- **Time:** O(1) for `get`, O(capacity) for evicting `put`
- **Space:** O(capacity)

## Approach 2: Frequency Buckets of LRU Lists (Optimal)
Maintain:
- `nodes`: key → node holding `(key, value, freq)`.
- `buckets`: freq → doubly linked list of nodes with that frequency, ordered by recency (front = most recent).
- `minFreq`: the smallest frequency currently present.

Touching a key removes its node from bucket `f` and appends it to the front of bucket `f + 1`; if bucket `f` becomes empty and `f == minFreq`, then `minFreq = f + 1`. Inserting a new key evicts the *tail* of bucket `minFreq` if full, then adds the new node to bucket 1 and sets `minFreq = 1`. Every step is a constant number of hash-map and linked-list operations.
- **Time:** O(1) per operation
- **Space:** O(capacity)

## Code (Optimal)

### C++
```cpp
#include <unordered_map>
#include <list>
using namespace std;

class LFUCache {
    struct Entry { int key, val, freq; };
    int cap, minFreq;
    unordered_map<int, list<Entry>::iterator> pos;    // key -> node
    unordered_map<int, list<Entry>> buckets;          // freq -> nodes (front = most recent)

    void touch(list<Entry>::iterator it) {
        Entry e = *it;
        buckets[e.freq].erase(it);
        if (buckets[e.freq].empty()) {
            buckets.erase(e.freq);
            if (minFreq == e.freq) ++minFreq;
        }
        ++e.freq;
        buckets[e.freq].push_front(e);
        pos[e.key] = buckets[e.freq].begin();
    }
public:
    LFUCache(int capacity) : cap(capacity), minFreq(0) {}

    int get(int key) {
        auto it = pos.find(key);
        if (it == pos.end()) return -1;
        int v = it->second->val;
        touch(it->second);
        return v;
    }

    void put(int key, int value) {
        if (cap == 0) return;
        auto it = pos.find(key);
        if (it != pos.end()) {
            it->second->val = value;
            touch(it->second);
            return;
        }
        if ((int)pos.size() == cap) {
            auto& lst = buckets[minFreq];
            pos.erase(lst.back().key);
            lst.pop_back();
            if (lst.empty()) buckets.erase(minFreq);
        }
        buckets[1].push_front({key, value, 1});
        pos[key] = buckets[1].begin();
        minFreq = 1;
    }
};
```

### Java
```java
import java.util.*;

class LFUCache {
    private static class Node {
        int key, val, freq = 1;
        Node(int k, int v) { key = k; val = v; }
    }

    private final int cap;
    private int minFreq = 0;
    private final Map<Integer, Node> nodes = new HashMap<>();
    // LinkedHashSet keeps insertion order: first = least recent within the bucket.
    private final Map<Integer, LinkedHashSet<Node>> buckets = new HashMap<>();

    public LFUCache(int capacity) { cap = capacity; }

    private void touch(Node n) {
        LinkedHashSet<Node> set = buckets.get(n.freq);
        set.remove(n);
        if (set.isEmpty()) {
            buckets.remove(n.freq);
            if (minFreq == n.freq) minFreq++;
        }
        n.freq++;
        buckets.computeIfAbsent(n.freq, f -> new LinkedHashSet<>()).add(n);
    }

    public int get(int key) {
        Node n = nodes.get(key);
        if (n == null) return -1;
        touch(n);
        return n.val;
    }

    public void put(int key, int value) {
        if (cap == 0) return;
        Node n = nodes.get(key);
        if (n != null) {
            n.val = value;
            touch(n);
            return;
        }
        if (nodes.size() == cap) {
            LinkedHashSet<Node> set = buckets.get(minFreq);
            Node victim = set.iterator().next();
            set.remove(victim);
            if (set.isEmpty()) buckets.remove(minFreq);
            nodes.remove(victim.key);
        }
        Node fresh = new Node(key, value);
        nodes.put(key, fresh);
        buckets.computeIfAbsent(1, f -> new LinkedHashSet<>()).add(fresh);
        minFreq = 1;
    }
}
```

### Python
```python
from collections import OrderedDict, defaultdict

class LFUCache:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.min_freq = 0
        self.vals = {}                              # key -> value
        self.freq = {}                              # key -> frequency
        self.buckets = defaultdict(OrderedDict)     # freq -> keys in recency order (first = oldest)

    def _touch(self, key: int) -> None:
        f = self.freq[key]
        del self.buckets[f][key]
        if not self.buckets[f]:
            del self.buckets[f]
            if self.min_freq == f:
                self.min_freq = f + 1
        self.freq[key] = f + 1
        self.buckets[f + 1][key] = None

    def get(self, key: int) -> int:
        if key not in self.vals:
            return -1
        self._touch(key)
        return self.vals[key]

    def put(self, key: int, value: int) -> None:
        if self.cap == 0:
            return
        if key in self.vals:
            self.vals[key] = value
            self._touch(key)
            return
        if len(self.vals) == self.cap:
            victim, _ = self.buckets[self.min_freq].popitem(last=False)
            if not self.buckets[self.min_freq]:
                del self.buckets[self.min_freq]
            del self.vals[victim]
            del self.freq[victim]
        self.vals[key] = value
        self.freq[key] = 1
        self.buckets[1][key] = None
        self.min_freq = 1
```

## Key Takeaways
- LFU = one LRU list per frequency plus a pointer to the minimum frequency; a touch just moves a node one bucket up.
- `minFreq` only ever needs to increase by one on a touch (the node came from the min bucket) or reset to 1 on an insert — no searching required.
- Handle `capacity == 0`, and evict *before* inserting the new key so the new key can never be the victim.
